/* ========================================================= IN-GAME AUTO UPDATER ========================================================= */ const { ipcRenderer } = require("electron"); const updateOverlay = document.getElementById("update-overlay"); const updateVersion = document.getElementById("update-version"); const updateNotes = document.getElementById("update-notes"); const updateStatus = document.getElementById("update-status"); const updateAction = document.getElementById("update-action"); const updateProgressContainer = document.getElementById("update-progress-container"); const updateProgressBar = document.getElementById("update-progress-bar"); let updateAvailable = false; function openUpdateModal(data) { updateAvailable = true; updateVersion.textContent = `Version ${data.currentVersion} → ${data.newVersion}`; updateNotes.textContent = data.releaseNotes || "No release notes provided."; updateStatus.textContent = "A new version of the game is available."; updateAction.textContent = "UPDATE NOW"; updateAction.disabled = false; updateProgressContainer.style.display = "none"; updateProgressBar.style.width = "0%"; updateOverlay.classList.add("open"); } updateAction.addEventListener("click", () => { if (!updateAvailable) { return; } updateAction.disabled = true; updateAction.textContent = "DOWNLOADING..."; updateStatus.textContent = "Downloading the update. Please wait..."; updateProgressContainer.style.display = "block"; ipcRenderer.send("download-update"); }); ipcRenderer.on("update-available", (event, data) => { openUpdateModal(data); }); ipcRenderer.on("update-progress", (event, progress) => { const percent = Math.max( 0, Math.min(100, progress.percent || 0) ); updateProgressBar.style.width = `${percent}%`; updateStatus.textContent = `Downloading update... ${percent}%`; }); ipcRenderer.on("update-downloaded", () => { updateProgressBar.style.width = "100%"; updateStatus.textContent = "Update downloaded. Restarting the game..."; updateAction.textContent = "RESTARTING..."; updateAction.disabled = true; }); ipcRenderer.on("post-update", (event, data) => { updateAvailable = false; updateVersion.textContent = `Updated to version ${data.version}`; updateNotes.textContent = data.releaseNotes || "No release notes provided."; updateStatus.textContent = "The update has been installed successfully."; updateAction.textContent = "CONTINUE"; updateAction.disabled = false; updateProgressContainer.style.display = "none"; updateProgressBar.style.width = "100%"; updateOverlay.classList.add("open"); updateAction.onclick = () => { updateOverlay.classList.remove("open"); updateAction.onclick = null; }; }); ipcRenderer.on("update-error", (event, data) => { updateStatus.textContent = "The update failed to download."; updateAction.textContent = "RETRY UPDATE"; updateAction.disabled = false; updateProgressContainer.style.display = "none"; updateAction.onclick = () => { updateAction.disabled = true; updateAction.textContent = "CHECKING..."; updateStatus.textContent = "Trying again..."; ipcRenderer.send("retry-update"); // Restore normal click handler after retry starts setTimeout(() => { updateAction.onclick = null; }, 100); }; });
